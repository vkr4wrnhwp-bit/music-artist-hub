# Audio Intelligence. Pinned, unlike everything above it, on purpose: this is
# a fast-moving vendor SDK and audio_elevenlabs.py is written against 2.65.0's
# real method signatures. Unpinned, a breaking major lands on the next deploy
# with no commit of ours to blame it on.
#
# The `pyaudio` extra is deliberately NOT requested - it is the only piece of
# the tree that needs a C toolchain, it is used for local microphone playback
# which the server has no use for, and asking for it would make the Render
# build depend on PortAudio headers being present.
#
# Optional at runtime. audio_providers.bootstrap() catches an ImportError here
# and leaves the mock adapters registered, so an install without this line
# still boots and still demos - it just never reaches a vendor.
elevenlabs==2.65.0

# Contract reading (2026-09-14): the text out of an uploaded PDF so the
# renewal dates can be found and checked. Pure Python, no C toolchain.
# Pinned for the same reason as elevenlabs. Optional at runtime:
# contract_reader says "cannot read PDFs yet" without it and the app
# still boots.
pypdf==6.18.1

# Error reporting (2026-09-19). The owner had set SENTRY_DSN in Render and
# nothing read it. observability.py reads it; with the variable unset the
# import is never attempted, so the app boots without this line too.
# Pinned like the two above: 2.69.2 is the newest release on PyPI today
# (checked with `pip index versions sentry-sdk`). The [flask] extra pulls
# blinker and the Flask integration, nothing with a C toolchain.
sentry-sdk[flask]==2.69.2
